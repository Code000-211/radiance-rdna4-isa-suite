cmake_minimum_required(VERSION 3.21)
project(rad_planner_cpu LANGUAGES CXX)
if(NOT EXISTS "${RADIANCE_SOURCE}/abi/rad_plugin.h")
  message(FATAL_ERROR "Set RADIANCE_SOURCE to the supplied Radiance checkout")
endif()
add_library(planner_cpu MODULE cpu_candidates.cpp)
target_include_directories(planner_cpu PRIVATE "${RADIANCE_SOURCE}/abi")
target_compile_features(planner_cpu PRIVATE cxx_std_17)
target_compile_options(planner_cpu PRIVATE -O3 -ffp-contract=off -fno-fast-math -Wall -Wextra)
set_target_properties(planner_cpu PROPERTIES PREFIX "" OUTPUT_NAME "planner_cpu")
