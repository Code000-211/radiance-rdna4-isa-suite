cmake_minimum_required(VERSION 3.21)
project(rad_replay_host_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_DIRECTORY "${RADIANCE_SOURCE}/core")
  message(FATAL_ERROR "Set -DRADIANCE_SOURCE=/path/to/radiance")
endif()
option(RAD_REPLAY_HOST_ASAN "Instrument host worker and test plugins; never GPU evidence" OFF)
if(RAD_REPLAY_HOST_ASAN)
  add_compile_options(-fsanitize=address -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address)
endif()
find_package(Threads REQUIRED)
set(R "${RADIANCE_SOURCE}")
add_library(replay-host-common STATIC
  ${R}/core/util/rad_util.cpp
  ${R}/core/device/dispatch.cpp ${R}/core/device/host.cpp
  ${R}/core/plugin/loader.cpp ${R}/core/plugin/schema.cpp ${R}/core/plugin/select.cpp
  ${R}/tools/kbench_core.cpp ${R}/core/format/encoding.cpp)
target_include_directories(replay-host-common PRIVATE ${R}/core ${R}/abi ${R}/tools ${R}/vendor)
target_compile_options(replay-host-common PRIVATE -ffunction-sections -fdata-sections -Wall -Wextra -Wno-unused-parameter)

foreach(tool rad-kbench-replay replay-capture-harness)
  if(tool STREQUAL "rad-kbench-replay")
    set(src ../worker.cpp)
  else()
    set(src ../../replay_tests/capture_harness.cpp)
  endif()
  add_executable(${tool} ${src})
  target_include_directories(${tool} PRIVATE .. ${R}/core ${R}/abi ${R}/tools ${R}/vendor)
  target_compile_options(${tool} PRIVATE -ffunction-sections -fdata-sections -Wall -Wextra -Wno-unused-parameter)
  target_link_options(${tool} PRIVATE -Wl,--gc-sections -Wl,--export-dynamic-symbol=rad_*)
  target_link_libraries(${tool} PRIVATE replay-host-common Threads::Threads ${CMAKE_DL_LIBS})
endforeach()
add_library(replay-test-plugin MODULE ../../replay_tests/probe_plugin.cpp)
target_include_directories(replay-test-plugin PRIVATE ${R}/abi)
target_link_libraries(replay-test-plugin PRIVATE Threads::Threads)
set_target_properties(replay-test-plugin PROPERTIES PREFIX "" OUTPUT_NAME "replay_test")

# Genuine libref built independently of the incomplete serving-engine source tree.
file(GLOB REF_SOURCES CONFIGURE_DEPENDS "${R}/libref/*.cpp")
add_library(replay-libref MODULE ${REF_SOURCES})
target_include_directories(replay-libref PRIVATE ${R}/abi)
target_compile_options(replay-libref PRIVATE -ffp-contract=off)
set_target_properties(replay-libref PROPERTIES PREFIX "" OUTPUT_NAME "libref")
add_executable(rad-replay-inventory ../inventory.cpp)
target_include_directories(rad-replay-inventory PRIVATE .. ${R}/core ${R}/abi ${R}/tools ${R}/vendor)
target_compile_options(rad-replay-inventory PRIVATE -ffunction-sections -fdata-sections)
target_link_options(rad-replay-inventory PRIVATE -Wl,--gc-sections -Wl,--export-dynamic-symbol=rad_*)
target_link_libraries(rad-replay-inventory PRIVATE replay-host-common Threads::Threads ${CMAKE_DL_LIBS})
